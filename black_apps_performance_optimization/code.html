// 1. Pinpointing Deadlocks and Victim AL Sessions during order processing spikes
traces
| where timestamp > ago(24h)
| where customDimensions.eventId == 'RT0012' // Database deadlock detected
| extend alStackTrace = tostring(customDimensions.alStackTrace),
        victimSession = tostring(customDimensions.sessionId),
        blockedTable = tostring(customDimensions.tableName)
| summarize DeadlockCount = count(), LastOccurrence = max(timestamp) by blockedTable, alStackTrace
| order by DeadlockCount desc

// 2. Profiling Long Running Queries exceeding 1000ms SLA in Core Ledger tables
traces
| where timestamp > ago(12h)
| where customDimensions.eventId == 'RT0005' // Slow SQL query
| extend executionTimeMs = todouble(customDimensions.executionTime),
        sqlStatement = tostring(customDimensions.sqlStatement),
        extensionName = tostring(customDimensions.extensionName)
| where executionTimeMs > 1200
| project timestamp, executionTimeMs, extensionName, sqlStatement
| top 5 by executionTimeMs desc
warning INDEX SCAN DETECTED: Query duration: 3,420 ms
[HIGH LATENCY] Table 17 'G/L Entry' unindexed filter on custom Dimension Set ID combination.
>> Recommended Fix: Added missing secondary key with Included fields on AL table extension, dropping query time from 3.42s to 12ms.
// 3. Isolating Job Queue starvation and background worker concurrency collisions
traces
| where timestamp > ago(6h)
| where customDimensions.eventId in ('RT0008', 'AL0000E26')

        failureReason = tostring(customDimensions.failureReason)
| summarize TotalFailures = count() by failureReason, jobQueueId
| order by TotalFailures desc
